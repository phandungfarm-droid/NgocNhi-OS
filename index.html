/* ========================================================================= CLOUD FUNCTIONS — HỆ THỐNG NGỌC NHI ------------------------------------------------------------------------- File này gồm 3 Cloud Function: 1) sendOrderAlertPush — GIỮ NGUYÊN 100% logic cũ (gửi push FCM thật khi có cảnh báo đơn hàng mới ở Quán ăn). Không sửa gì trong hàm này. 2) getR2VideoUploadUrl — MỚI. Cấp URL upload đã ký trước (presigned PUT URL) để client PUT thẳng file video lên Cloudflare R2, KHÔNG đi qua server này (server chỉ ký URL, không nhận file). Khoá bí mật R2 chỉ tồn tại ở đây (functions.config().r2.*), không lộ ra client. 3) deleteR2Video — MỚI. Xoá 1 object video trên R2 theo URL công khai hiện có (client gọi trước khi ghi đè/xoá tham chiếu trong Realtime Database), dùng khi "Đổi video" hoặc "Xoá video". CẤU HÌNH BẮT BUỘC trước khi deploy (chạy 1 lần, thay đúng giá trị của bạn): firebase functions:config:set ^ r2.account_id="XXXXXXXXXXXXXXXXXXXXXXXX" ^ r2.access_key_id="XXXXXXXXXXXXXXXXXXXX" ^ r2.secret_access_key="XXXXXXXXXXXXXXXXXXXXXXXXXXXXXXXX" ^ r2.bucket="ngocnhi-videos" ^ r2.public_base_url="https://videos.yourdomain.com" (public_base_url là domain công khai đã gắn cho bucket R2 — Custom Domain hoặc "https://pub-xxxxxxxx.r2.dev" nếu dùng public dev URL của Cloudflare — xem r2-setup-instructions.md đính kèm.) Sau khi set config, deploy lại: firebase deploy --only functions ========================================================================= */ const functions = require("firebase-functions/v1"); const admin = require("firebase-admin"); const { S3Client, PutObjectCommand, DeleteObjectCommand } = require("@aws-sdk/client-s3"); const { getSignedUrl } = require("@aws-sdk/s3-request-presigner"); admin.initializeApp(); /* ========================= 1) GIỮ NGUYÊN — KHÔNG SỬA ==================== */ exports.sendOrderAlertPush = functions .region("asia-southeast1") // trùng khu vực Realtime Database (asia-southeast1) — đổi nếu project bạn khác .database.ref("/Restaurant/OrderAlerts/{alertId}") .onCreate(async (snapshot, context) => { const alert = snapshot.val(); if (!alert || !alert.message) return null; const tokensSnap = await admin.database().ref("/Restaurant/NotificationTokens").once("value"); const tokensObj = tokensSnap.val() || {}; const tokens = Object.keys(tokensObj); if (tokens.length === 0) { console.log("Chưa có thiết bị nào đăng ký nhận thông báo (Restaurant/NotificationTokens rỗng)."); return null; } const payload = { notification: { title: alert.kind === "table" ? "Đơn mới tại bàn" : "Đơn online mới", body: alert.message }, data: { kind: alert.kind || "", alertId: context.params.alertId, orderId: alert.orderId || "", ticketId: alert.ticketId || "", tableId: alert.tableId || "" } }; const response = await admin.messaging().sendEachForMulticast({ tokens: tokens, notification: payload.notification, data: payload.data }); // Dọn các token không còn hợp lệ (thiết bị gỡ cài đặt/thu hồi quyền) — // chỉ xoá token hỏng trong NotificationTokens, KHÔNG đụng OrderAlerts. const updates = {}; response.responses.forEach((res, idx) => { if (!res.success) { const code = res.error && res.error.code; if (code === "messaging/registration-token-not-registered" || code === "messaging/invalid-registration-token") { updates["/Restaurant/NotificationTokens/" + tokens[idx]] = null; } } }); if (Object.keys(updates).length > 0) { await admin.database().ref().update(updates); } console.log(`Đã gửi push cho cảnh báo ${context.params.alertId}: ${response.successCount}/${tokens.length} thành công.`); return null; }); /* ========================= R2 — HÀM DÙNG CHUNG =========================== */ function getR2Client() { const cfg = functions.config().r2 || {}; if (!cfg.account_id || !cfg.access_key_id || !cfg.secret_access_key || !cfg.bucket || !cfg.public_base_url) { throw new functions.https.HttpsError( "failed-precondition", "Chưa cấu hình Cloudflare R2 đầy đủ. Chạy: firebase functions:config:set r2.account_id=... r2.access_key_id=... r2.secret_access_key=... r2.bucket=... r2.public_base_url=..." ); } return { client: new S3Client({ region: "auto", endpoint: `https://${cfg.account_id}.r2.cloudflarestorage.com`, credentials: { accessKeyId: cfg.access_key_id, secretAccessKey: cfg.secret_access_key } }), bucket: cfg.bucket, publicBaseUrl: String(cfg.public_base_url).replace(/\/+$/, "") }; } var R2_ALLOWED_MODULES = { farm: true, restaurant: true, wedding: true, admin: true }; var R2_ALLOWED_CONTENT_TYPES = { "video/mp4": "mp4", "video/webm": "webm" }; var R2_MAX_BYTES = 100 * 1024 * 1024; // 100MB — trùng giới hạn cũ đã áp dụng khi còn dùng Firebase Storage /* ===== 2) MỚI — Cấp URL upload đã ký trước (video module, chỉ Admin) ===== */ exports.getR2VideoUploadUrl = functions .region("asia-southeast1") .https.onCall(async (data, context) => { // Cùng quy tắc phân quyền hiện có của DA2: mọi tài khoản đăng nhập // Firebase Auth thành công đều được coi là Administrator. if (!context.auth) { throw new functions.https.HttpsError("unauthenticated", "Cần đăng nhập với tài khoản quản trị."); } const moduleKey = data && data.moduleKey; const contentType = data && data.contentType; if (!R2_ALLOWED_MODULES[moduleKey]) { throw new functions.https.HttpsError("invalid-argument", "Module không hợp lệ."); } const ext = R2_ALLOWED_CONTENT_TYPES[contentType]; if (!ext) { throw new functions.https.HttpsError("invalid-argument", "Định dạng video không hỗ trợ (chỉ MP4/WEBM)."); } const { client, bucket, publicBaseUrl } = getR2Client(); const key = `module-videos/${moduleKey}-${Date.now()}.${ext}`; const command = new PutObjectCommand({ Bucket: bucket, Key: key, ContentType: contentType, ContentLengthRange: undefined // R2/S3 presigned PUT không giới hạn size qua đây — giới hạn được kiểm tra ở client (R2_MAX_BYTES) trước khi PUT }); const uploadUrl = await getSignedUrl(client, command, { expiresIn: 300 }); // URL có hiệu lực 5 phút return { uploadUrl: uploadUrl, publicUrl: `${publicBaseUrl}/${key}`, key: key, maxBytes: R2_MAX_BYTES }; }); /* ===== 3) MỚI — Xoá 1 video khỏi R2 theo URL công khai (chỉ Admin) ===== */ exports.deleteR2Video = functions .region("asia-southeast1") .https.onCall(async (data, context) => { if (!context.auth) { throw new functions.https.HttpsError("unauthenticated", "Cần đăng nhập với tài khoản quản trị."); } const url = data && data.url; if (!url) return { deleted: false, reason: "no-url" }; const { client, bucket, publicBaseUrl } = getR2Client(); if (url.indexOf(publicBaseUrl + "/") !== 0) { // URL không thuộc đúng bucket/domain R2 đang cấu hình — không xoá gì, // tránh xoá nhầm dữ liệu ở nơi khác (ví dụ video nhập tay bằng URL lạ). return { deleted: false, reason: "url-mismatch" }; } const key = url.slice(publicBaseUrl.length + 1); try { await client.send(new DeleteObjectCommand({ Bucket: bucket, Key: key })); return { deleted: true, key: key }; } catch (err) { console.warn("Không xoá được video trên R2:", err && err.message); return { deleted: false, reason: err && err.message }; } });
